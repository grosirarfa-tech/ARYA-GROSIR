function doGet(e) {
  var action = e.parameter.action;
  if(action === "getData") {
    return ContentService.createTextOutput(JSON.stringify(getAllData()))
      .setMimeType(ContentService.MimeType.JSON);
  }
  return ContentService.createTextOutput("ARYA GROSIR Apps Script Ready");
}

function doPost(e) {
  try {
    var data = JSON.parse(e.postData.contents);
    saveAllData(data);
    return ContentService.createTextOutput(JSON.stringify({status: "success"}))
      .setMimeType(ContentService.MimeType.JSON);
  } catch(err) {
    return ContentService.createTextOutput(JSON.stringify({status: "error", message: err.toString()}))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

function getAllData() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  return {
    katalog: getSheetData(ss, "KATALOG BARANG"),
    belanja: getSheetData(ss, "BELANJA"),
    penjualan: getSheetData(ss, "PENJUALAN"),
    pengeluaran: getSheetData(ss, "PENGELUARAN LAIN")
  };
}
