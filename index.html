SINKRONISASI GOOGLE SPREADSHEET

Integrasi 2-Arah via Google Apps Script (Auto Backup 23.00)

Panduan Setup Google Apps Script Backend

  1. Buat Google Spreadsheet baru di Google Drive Anda.
  2. Buat sheet-sheet dengan nama persis sebagai berikut:
    • KATALOG BARANG
    • BELANJA
    • PENJUALAN
    • PERKIRAAN BELANJA
    • PENGELUARAN LAIN
    • KEUNTUNGAN HARIAN
    • KEUNTUNGAN PERBULAN
    • SINGKRONISASI
  3. Buka menu Ekstensi > Apps Script.
  4. Hapus semua kode lama, lalu salin dan tempelkan kode GAS di bawah ini:
/* GOOGLE APPS SCRIPT BACKEND "ARYA GROSIR" POS */
function doPost(e) {
  try {
    var data = JSON.parse(e.postData.contents);
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    
    // Save Katalog
    if(data.katalog) {
      var sheetKat = ss.getSheetByName('KATALOG BARANG');
      sheetKat.clearContents();
      sheetKat.appendRow(['BARCODE','NAMA BARANG','SATUAN','QTY','MINIMAL QTY','HARGA POKOK','HARGA JUAL GROSIR','HARGA JUAL RETAIL']);
      data.katalog.forEach(function(r) {
        sheetKat.appendRow([r.barcode, r.nama, r.satuan, r.qty, r.minQty, r.hargaPokok, r.hargaGrosir, r.hargaRetail]);
      });
    }

    // Save Belanja
    if(data.belanja) {
      var sheetBlj = ss.getSheetByName('BELANJA');
      sheetBlj.clearContents();
      sheetBlj.appendRow(['ID','TANGGAL','BARCODE','NAMA BARANG','QTY','SATUAN','TOTAL HARGA PER ITEM','TOTAL HARGA']);
      data.belanja.forEach(function(b) {
        b.items.forEach(function(item) {
          sheetBlj.appendRow([b.id, b.tanggal, item.barcode, item.nama, item.qty, item.satuan, item.totalHargaItem, b.totalHarga]);
        });
      });
    }

    // Save Penjualan
    if(data.penjualan) {
      var sheetPjl = ss.getSheetByName('PENJUALAN');
      sheetPjl.clearContents();
      sheetPjl.appendRow(['ID','TANGGAL','TYPE PEMBELI','NAMA PEMBELI','BARCODE','NAMA BARANG','SATUAN','QTY','TOTAL HARGA PER ITEM','TOTAL HARGA']);
      data.penjualan.forEach(function(p) {
        p.items.forEach(function(item) {
          sheetPjl.appendRow([p.id, p.tanggal, p.typePembeli, p.namaPembeli, item.barcode, item.nama, item.satuan, item.qty, item.totalHargaItem, p.totalHarga]);
        });
      });
    }

    // Save Pengeluaran
    if(data.pengeluaran) {
      var sheetExp = ss.getSheetByName('PENGELUARAN LAIN');
      sheetExp.clearContents();
      sheetExp.appendRow(['TANGGAL','KETERANGAN','JUMLAH']);
      data.pengeluaran.forEach(function(ex) {
        sheetExp.appendRow([ex.tanggal, ex.keterangan, ex.jumlah]);
      });
    }

    // Record Sync Log
    var sheetSync = ss.getSheetByName('SINGKRONISASI');
    sheetSync.appendRow([new Date().toLocaleString('id-ID'), 'BERHASIL BACKUP MANUAL/AUTO']);

    return ContentService.createTextOutput(JSON.stringify({status: 'SUCCESS'}))
      .setMimeType(ContentService.MimeType.JSON);
  } catch(err) {
    return ContentService.createTextOutput(JSON.stringify({status: 'ERROR', message: err.toString()}))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
